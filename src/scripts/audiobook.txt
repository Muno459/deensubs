(function(){
'use strict';
var SLUG='__SLUG__',TITLE='__TITLE__',ART='__ART__',SCHOLAR='__SCHOLAR__',DUR=+'__DUR__'||0;
var chapters=__CHAPTERS__;
var au=document.getElementById('ab-audio');
var $=function(id){return document.getElementById(id)};
var playBtn=$('ab-play'),back=$('ab-back'),fwd=$('ab-fwd'),seek=$('ab-seek'),cur=$('ab-cur'),tot=$('ab-tot'),
    speedBtn=$('ab-speed'),tr=$('ab-tr'),arBtn=$('ab-arbtn'),chLabel=$('ab-chlabel'),followBtn=$('ab-follow'),
    trwrap=$('ab-trwrap'),stage=$('abv'),fsBtn=$('ab-fs'),bg=$('abv-bg'),shareBtn=$('ab-share'),
    chrRail=$('chr-ls');
if(!au)return;

function fmt(s){s=Math.max(0,Math.floor(s));var h=Math.floor(s/3600),m=Math.floor(s%3600/60),x=s%60;
return (h?h+':'+String(m).padStart(2,'0'):m)+':'+String(x).padStart(2,'0')}
tot.textContent=fmt(DUR);

/* trust the decoder over the DB — covers rows published with duration 0 */
au.addEventListener('loadedmetadata',function(){
  if(au.duration&&isFinite(au.duration)&&Math.abs(au.duration-DUR)>1){
    DUR=au.duration;seek.max=DUR;tot.textContent=fmt(DUR);renderTicks();
  }
});

/* the element that actually scrolls the transcript */
function scroller(){
  return stage?trwrap:tr; // classic: .ab-tr is its own scroll box
}
function centerActive(){
  var el=unitEls[active];if(!el)return;
  var sc=scroller(),r=el.getBoundingClientRect(),s=sc.getBoundingClientRect();
  sc.scrollTo({top:sc.scrollTop+(r.top-s.top)-(sc.clientHeight/2-r.height/2),behavior:'smooth'});
}

/* ---- transcript ---- */
var words=[],units=[],unitEls=[],active=-1,follow=true,arOn=false;
fetch((window.__CDN||'https://cdn.deensubs.com')+'/transcripts/'+SLUG+'.json?v='+new Date().toISOString().slice(0,10))
.then(function(r){if(!r.ok)throw 0;return r.json()})
.then(function(doc){
  words=doc.words||[];
  units=(doc.units||[]).map(function(u){
    var w0=u[1],w1=u[2];
    return {t:u[0],w0:w0,w1:w1,a:words[w0]?words[w0][1]:0,b:words[w1]?words[w1][2]:0};
  });
  var html='';
  (doc.paragraphs&&doc.paragraphs.length?doc.paragraphs:[[0,units.length-1]]).forEach(function(p){
    html+='<p class="ab-p">';
    for(var i=p[0];i<=p[1]&&i<units.length;i++){
      html+='<span class="ab-u" data-i="'+i+'"><span class="ab-en">'+esc(units[i].t)+'</span></span> ';
    }
    html+='</p>';
  });
  tr.innerHTML=html;
  unitEls=tr.querySelectorAll('.ab-u');
  tr.addEventListener('click',function(ev){
    var el=ev.target.closest('.ab-u');if(!el)return;
    var u=units[+el.dataset.i];
    au.currentTime=u.a+0.01;au.play();
  });
})
.catch(function(){tr.innerHTML='<p class="ab-p ab-dim">Transcript unavailable.</p>'});
function esc(s){return s.replace(/&/g,'&amp;').replace(/</g,'&lt;')}

/* Arabic toggle: built lazily from the shared word array */
if(arBtn)arBtn.addEventListener('click',function(){
  arOn=!arOn;arBtn.classList.toggle('on',arOn);
  if(arOn&&!tr.querySelector('.ab-ar')){
    unitEls.forEach(function(el){
      var u=units[+el.dataset.i],ar=[];
      for(var k=u.w0;k<=u.w1&&k<words.length;k++)ar.push(words[k][0]);
      var sp=document.createElement('span');sp.className='ab-ar';sp.dir='rtl';sp.textContent=ar.join(' ');
      el.appendChild(sp);
    });
  }
  tr.classList.toggle('ab-showar',arOn);
});

/* ---- karaoke sync ---- */
function findUnit(t){ // raw last-started unit; the silence-gap rule lives in tick
  var lo=0,hi=units.length-1,ans=-1;
  while(lo<=hi){var mid=(lo+hi)>>1;
    if(units[mid].a<=t){ans=mid;lo=mid+1}else hi=mid-1}
  return ans;
}
var wordPtr=0,enWords=null,litCount=0;
function splitEn(el){
  // lazily split the active unit's English into word spans (kept afterwards)
  var en=el.querySelector('.ab-en');
  if(!en)return null;
  if(!en.dataset.split){
    var parts=en.textContent.split(/\s+/).filter(Boolean);
    en.innerHTML=parts.map(function(w){return '<i class="ab-w">'+esc(w)+'</i>'}).join(' ');
    en.dataset.split='1';
  }
  return en.querySelectorAll('.ab-w');
}
/* Apple-style read state: everything already spoken keeps a soft white */
var wasTo=-1;
function markRead(target){
  if(target===wasTo)return;
  var k;
  if(target>wasTo){for(k=wasTo+1;k<=target;k++)if(unitEls[k])unitEls[k].classList.add('was')}
  else for(k=target+1;k<=wasTo;k++)if(unitEls[k])unitEls[k].classList.remove('was');
  wasTo=target;
}
function sweep(u,t){
  // Arabic-word-anchored progress, quantized to ENGLISH word boundaries —
  // a continuous gradient stopped mid-glyph, whole words never do
  if(!enWords)return;
  var span=u.w1-u.w0+1,k=Math.max(u.w0,Math.min(wordPtr,u.w1));
  while(k<u.w1&&words[k+1]&&words[k+1][1]<=t)k++;
  while(k>u.w0&&words[k]&&words[k][1]>t)k--;
  wordPtr=k;
  var w=words[k],frac=w?Math.max(0,Math.min(1,(t-w[1])/Math.max(0.05,w[2]-w[1]))):0;
  var p=Math.max(0,Math.min(1,(k-u.w0+frac)/span));
  var lit=Math.round(p*enWords.length);
  if(lit===litCount)return;
  for(var i=Math.min(lit,litCount);i<Math.max(lit,litCount);i++){
    if(enWords[i])enWords[i].classList.toggle('on',i<lit);
  }
  litCount=lit;
}
function tick(){
  var t=au.currentTime;
  if(units.length){
    var raw=findUnit(t);
    var i=raw>=0&&t>units[raw].b+2&&raw<units.length-1?-1:raw; // silence gap
    markRead(i>=0?i-1:(raw>=0&&t>units[raw].b?raw:raw-1));
    if(i!==active){
      if(active>=0&&unitEls[active]){
        unitEls[active].classList.remove('on');
        if(enWords)enWords.forEach(function(w){w.classList.remove('on')});
      }
      active=i;enWords=null;litCount=0;
      if(i>=0&&unitEls[i]){
        unitEls[i].classList.add('on');wordPtr=units[i].w0;
        enWords=splitEn(unitEls[i]);
        if(follow)centerActive();
      }
    }
    if(active>=0)sweep(units[active],t);
  }
  if(!seek.dragging)seek.value=t;
  cur.textContent=fmt(t);
  updChapter(t);
  if(!au.paused)requestAnimationFrame(tick);
}
au.addEventListener('play',function(){playBtn.classList.add('playing');if(stage)stage.classList.add('playing');requestAnimationFrame(tick)});
au.addEventListener('pause',function(){playBtn.classList.remove('playing');if(stage)stage.classList.remove('playing')});
au.addEventListener('timeupdate',function(){if(au.paused)tick()});

/* follow mode: manual scrolling pauses auto-follow */
['wheel','touchmove'].forEach(function(evt){
  trwrap.addEventListener(evt,function(){follow=false;followBtn.hidden=false},{passive:true});
});
followBtn.addEventListener('click',function(){
  follow=true;followBtn.hidden=true;
  if(active>=0&&unitEls[active])centerActive();
});

/* ---- controls ---- */
playBtn.addEventListener('click',function(){au.paused?au.play():au.pause()});
back.addEventListener('click',function(){au.currentTime=Math.max(0,au.currentTime-15)});
fwd.addEventListener('click',function(){au.currentTime=Math.min(DUR,au.currentTime+15)});
seek.max=DUR||100;
seek.addEventListener('input',function(){seek.dragging=true;cur.textContent=fmt(+seek.value)});
seek.addEventListener('change',function(){au.currentTime=+seek.value;seek.dragging=false;if(follow&&active>=0)centerActive()});
var speeds=[1,1.25,1.5,1.75,2],si=0;
speedBtn.addEventListener('click',function(){
  si=(si+1)%speeds.length;au.playbackRate=speeds[si];
  speedBtn.textContent=speeds[si]+'×';
});

/* stage extras: tap artwork to play, double-tap / f / button for fullscreen */
if(bg){
  bg.addEventListener('click',function(){au.paused?au.play():au.pause()});
  bg.addEventListener('dblclick',function(){toggleFs()});
}
function toggleFs(){
  if(!stage)return;
  if(document.fullscreenElement){document.exitFullscreen();return}
  if(stage.requestFullscreen)stage.requestFullscreen();
  else if(stage.webkitRequestFullscreen)stage.webkitRequestFullscreen();
  else{
    // iPhone Safari: no element fullscreen — CSS theater mode instead
    var on=stage.classList.toggle('fakefs');
    document.documentElement.classList.toggle('ab-noscroll',on);
    if(follow&&active>=0)centerActive();
  }
}
if(fsBtn){
  fsBtn.addEventListener('click',toggleFs);
  document.addEventListener('fullscreenchange',function(){stage.classList.toggle('fs',document.fullscreenElement===stage)});
}
if(shareBtn)shareBtn.addEventListener('click',function(){
  var url=location.origin+'/watch/'+SLUG;
  if(navigator.share){navigator.share({title:TITLE,url:url}).catch(function(){})}
  else if(navigator.clipboard){
    navigator.clipboard.writeText(url);
    var sp=shareBtn.querySelector('span'),old=sp.textContent;
    sp.textContent='Copied!';setTimeout(function(){sp.textContent=old},1600);
  }
});
document.addEventListener('keydown',function(ev){
  if(/INPUT|TEXTAREA/.test(document.activeElement.tagName))return;
  if(ev.code==='Space'){ev.preventDefault();playBtn.click()}
  else if(ev.key==='ArrowLeft')back.click();
  else if(ev.key==='ArrowRight')fwd.click();
  else if((ev.key==='f'||ev.key==='F')&&stage)toggleFs();
});

/* chapters */
var chrCards=chrRail?chrRail.querySelectorAll('.chr-c'):null;
if(chrRail)chrRail.addEventListener('click',function(ev){
  var c=ev.target.closest('.chr-c');if(!c)return;
  au.currentTime=+c.dataset.t;au.play();
});
function updChapter(t){
  if(!chapters.length||!chLabel)return;
  var ci=-1;
  for(var i=0;i<chapters.length;i++)if(chapters[i].t<=t)ci=i;
  chLabel.textContent=ci>=0?chapters[ci].title:'';
  if(chrCards)for(var k=0;k<chrCards.length;k++)chrCards[k].classList.toggle('chr-on',k===ci);
}
function renderTicks(){
  var ticks=$('ab-ticks');
  if(!ticks||!chapters.length||!DUR)return;
  ticks.innerHTML='';
  chapters.forEach(function(c){
    if(!c.t)return;
    var el=document.createElement('i');el.style.left=(c.t/DUR*100)+'%';ticks.appendChild(el);
  });
}
renderTicks();

/* lock-screen metadata (Apple-Podcasts feel) */
if('mediaSession' in navigator){
  navigator.mediaSession.metadata=new MediaMetadata({
    title:TITLE,artist:SCHOLAR||'DeenSubs',album:'DeenSubs',
    artwork:ART?[{src:ART,sizes:'640x640',type:'image/webp'}]:[]
  });
  navigator.mediaSession.setActionHandler('seekbackward',function(){back.click()});
  navigator.mediaSession.setActionHandler('seekforward',function(){fwd.click()});
  navigator.mediaSession.setActionHandler('play',function(){au.play()});
  navigator.mediaSession.setActionHandler('pause',function(){au.pause()});
}
})();

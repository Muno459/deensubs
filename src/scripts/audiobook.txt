}

function findUnit(t){ // raw last-started unit; the silence-gap rule lives in tick
  var lo=0,hi=units.length-1,ans=-1;
  while(lo<=hi){var mid=(lo+hi)>>1;
    if(units[mid].a<=t){ans=mid;lo=mid+1}else hi=mid-1}
  return ans;
}
var wordPtr=0,enWords=null,litCount=0,anch=null;
/* Transliteration anchors: names and Islamic terms appear in BOTH texts
   (al-Fida' <-> \u0627\u0644\u0641\u062f\u0627\u0621). Matching consonant skeletons pins the
   sweep to reality at those words; between anchors it interpolates. */
var AR={'\u0627':'a','\u0623':'a','\u0625':'a','\u0622':'a','\u0628':'b','\u062a':'t','\u062b':'t','\u062c':'j','\u062d':'h','\u062e':'k','\u062f':'d','\u0630':'d','\u0631':'r','\u0632':'z','\u0633':'s','\u0634':'s','\u0635':'s','\u0636':'d','\u0637':'t','\u0638':'z','\u0639':'','\u063a':'g','\u0641':'f','\u0642':'q','\u0643':'k','\u0644':'l','\u0645':'m','\u0646':'n','\u0647':'h','\u0648':'w','\u064a':'y','\u0649':'a','\u0629':'h','\u0621':'','\u0626':'','\u0624':''};
function skel(t){
  t=(t||'').toLowerCase().replace(/[\u064b-\u0652\u0670\u0640]/g,'');
  t=t.replace(/^\u0627\u0644/,'').replace(/^al[-\u2019']?/,'');
  var o='';
  for(var i=0;i<t.length;i++){var ch=t[i];o+=(AR[ch]!==undefined?AR[ch]:ch)}
  return o.replace(/[^a-z]/g,'').replace(/[aeiouwhy]/g,'');
}
function buildAnchors(u,parts){
  var A=[],lastK=u.w0-1;
  for(var e=0;e<parts.length;e++){
    var es=skel(parts[e]);
    if(es.length<2)continue;
    for(var k=lastK+1;k<=u.w1&&k<words.length;k++){
      var as=skel(words[k][0]);
      if(as.length<2)continue;
      var hit=(as===es)||(es.length>2&&as.length>2&&(as.indexOf(es)>=0||es.indexOf(as)>=0));
      if(hit){A.push([k-u.w0,e]);lastK=k;break;}
    }
  }
  return A;
}
function splitEn(el){
  // lazily split a unit's English into word spans (kept afterwards),
  // walking the tree so <b>/<i> markup survives the split
  var en=el.querySelector('.ab-en');
  if(!en)return null;
  if(!en.dataset.split){
    (function wrap(node){
      for(var c=node.firstChild;c;){
        var nx=c.nextSibling;
        if(c.nodeType===3){
          var parts=c.textContent.split(/(\s+)/),frag=document.createDocumentFragment();
          for(var j=0;j<parts.length;j++){
            if(!parts[j])continue;
            if(/^\s+$/.test(parts[j]))frag.appendChild(document.createTextNode(parts[j]));
            else{var sp=document.createElement('span');sp.className='ab-w';sp.textContent=parts[j];frag.appendChild(sp)}
          }
          node.replaceChild(frag,c);
        }else if(c.nodeType===1&&!/\bab-(w|ar)\b/.test(c.className))wrap(c);
        c=nx;
      }
    })(en);
    en.dataset.split='1';
  }
  return en.querySelectorAll('.ab-w');
}
/* Apple-style read state: everything already spoken keeps a soft white */
var wasTo=-1;
function markRead(target){
  if(target===wasTo)return;
  var k;
  if(target>wasTo){for(k=wasTo+1;k<=target;k++)if(unitEls[k])unitEls[k].classList.add('was')}
  else for(k=target+1;k<=wasTo;k++)if(unitEls[k])unitEls[k].classList.remove('was');
  wasTo=target;
}
function sweep(u,t){
  // Arabic-word-anchored progress, quantized to ENGLISH word boundaries.
  // Transliteration anchors pin names; interpolation runs between them.
  if(!enWords)return;
  var span=u.w1-u.w0+1,k=Math.max(u.w0,Math.min(wordPtr,u.w1));
  while(k<u.w1&&words[k+1]&&words[k+1][1]<=t)k++;
  while(k>u.w0&&words[k]&&words[k][1]>t)k--;
  wordPtr=k;
  var w=words[k],frac=w?Math.max(0,Math.min(1,(t-w[1])/Math.max(0.05,w[2]-w[1]))):0;
  var pos=k-u.w0+frac,lit;
  if(anch&&anch.length&&typeof anch[0]==='number'){
    // complete word map: sweep through the group across the word's duration
    var gk=Math.max(0,Math.min(anch.length-1,k-u.w0)),g0=gk?anch[gk-1]:0;
    lit=g0+Math.ceil(frac*(anch[gk]-g0));
    if(anch[gk]>g0&&lit===g0)lit=g0+1;
  }else if(anch&&anch.length){
    var pts=[[0,0]].concat(anch).concat([[span,enWords.length]]);
    var lo=pts[0],hi=pts[pts.length-1];
    for(var i=0;i<pts.length-1;i++){
      if(pts[i][0]<=pos&&pos<=pts[i+1][0]){lo=pts[i];hi=pts[i+1];break}
    }
    var f=hi[0]>lo[0]?(pos-lo[0])/(hi[0]-lo[0]):0;
    lit=Math.round(lo[1]+f*(hi[1]-lo[1])+0.5);
  }else{
    lit=Math.round(Math.max(0,Math.min(1,pos/span))*enWords.length);
  }
  lit=Math.max(0,Math.min(enWords.length,lit));
  if(lit===litCount)return;
  for(var i=Math.min(lit,litCount);i<Math.max(lit,litCount);i++){
    if(enWords[i])enWords[i].classList.toggle('on',i<lit);
  }
  litCount=lit;
}
function tick(){
  var t=au.currentTime;
  if(units.length){
    var raw=findUnit(t);
    var i=raw>=0&&t>units[raw].b+2&&raw<units.length-1?-1:raw; // silence gap
    markRead(i>=0?i-1:(raw>=0&&t>units[raw].b?raw:raw-1));
    if(i!==active){
      if(active>=0&&unitEls[active]){
        unitEls[active].classList.remove('on');
        if(enWords)enWords.forEach(function(w){w.classList.remove('on')});
      }
      active=i;enWords=null;litCount=0;anch=null;
      if(i>=0&&unitEls[i]){
        unitEls[i].classList.add('on');wordPtr=units[i].w0;
        enWords=splitEn(unitEls[i]);
        if(enWords){
          var pw=[];enWords.forEach(function(x){pw.push(x.textContent)});
          anch=(alignPairs&&alignPairs[i]&&alignPairs[i].length)?alignPairs[i]:buildAnchors(units[i],pw);
        }
        if(follow)centerActive();
      }
    }
    if(active>=0)sweep(units[active],t);
  }
  if(!seek.dragging)seek.value=t;
  cur.textContent=fmt(t);
  if(pctEl&&DUR){
    var pc=Math.max(0,Math.min(100,t/DUR*100));
    pctEl.textContent=Math.round(pc)+'%';
    if(progEl)progEl.style.width=pc+'%';
  }
  updChapter(t);
  if(!au.paused)requestAnimationFrame(tick);
}

function updChapter(t){
  if(!chapters.length)return;
  var ci=-1,i;
  for(i=0;i<chapters.length;i++)if(chapters[i].t<=t)ci=i;
  var di=ci<0?0:ci; // before the first mark the reader is still in chapter 1
  for(i=0;i<chlT.length;i++)chlT[i].textContent=chapters[di].title;
  for(i=0;i<chlN.length;i++)chlN[i].textContent='Chapter '+(di+1);
  if(chrCards)for(i=0;i<chrCards.length;i++)chrCards[i].classList.toggle('chr-on',i===ci);
  if(chiEls)for(i=0;i<chiEls.length;i++)chiEls[i].classList.toggle('chi-on',i===ci);
}
function renderTicks(){
  var ticks=$('ab-ticks');
  if(!ticks||!chapters.length||!DUR)return;
  ticks.innerHTML='';
  chapters.forEach(function(c){
    if(!c.t)return;
    var el=document.createElement('i');el.style.left=(c.t/DUR*100)+'%';ticks.appendChild(el);
  });
}

  navigator.mediaSession.setActionHandler('pause',function(){au.pause()});
  try{
    navigator.mediaSession.setActionHandler('seekto',function(d){
      if(d.fastSeek&&au.fastSeek){au.fastSeek(d.seekTime);return}
      au.currentTime=d.seekTime;
    });
  }catch(e){}
  // real progress bar + speed on the lock screen
  var msPos=function(){
    if(!DUR||!isFinite(DUR))return;
    try{navigator.mediaSession.setPositionState({duration:DUR,playbackRate:au.playbackRate,position:Math.min(au.currentTime,DUR)})}catch(e){}
  };
  au.addEventListener('timeupdate',function(){if(Math.floor(au.currentTime)%5===0)msPos()});
  au.addEventListener('ratechange',msPos);
  au.addEventListener('seeked',msPos);
}

/* ---- sleep timer: moon button cycles Off / 15 / 30 / 45 / 60 min ---- */
var slBtn=$('ab-sleep');
if(slBtn){
  var slSteps=[0,15,30,45,60],slIdx=0,slDeadline=0,slTick=null;
  var slPill=function(msg){
    var p=document.createElement('div');p.className='ab-resumed';p.textContent=msg;
    document.body.appendChild(p);
    setTimeout(function(){p.classList.add('out');setTimeout(function(){p.remove()},450)},2200);
  };
  slBtn.addEventListener('click',function(){
    slIdx=(slIdx+1)%slSteps.length;
    clearInterval(slTick);slTick=null;
    if(!slSteps[slIdx]){
      slBtn.classList.remove('on');slBtn.textContent='☾';
      slPill('Sleep timer off');return;
    }
    slDeadline=performance.now()+slSteps[slIdx]*60000;
    slBtn.classList.add('on');
    slPill('Sleep in '+slSteps[slIdx]+' min');
    slTick=setInterval(function(){
      var left=slDeadline-performance.now();
      if(left<=0){
        clearInterval(slTick);slTick=null;slIdx=0;
        slBtn.classList.remove('on');slBtn.textContent='☾';
        au.pause();slPill('Sleep timer — paused');return;
      }
      slBtn.textContent=Math.ceil(left/60000)+'m';
    },5000);
    slBtn.textContent=slSteps[slIdx]+'m';
  });
}
})();
